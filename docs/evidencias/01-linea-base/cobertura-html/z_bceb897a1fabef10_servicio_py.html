1"""Servicio de estimación de duración de procedimientos. 

2 

3Este módulo es el punto de integración del componente predictivo con el resto 

4del sistema. Aplica una cadena de respaldo de tres niveles, de modo que la 

5agenda siempre obtiene una estimación utilizable aunque el modelo de 

6aprendizaje automático todavía no esté entrenado: 

7 

8 1. MODELO — si existe una versión activa del modelo entrenado, se usa. 

9 2. HISTÓRICO — si no, se promedia la duración real de las citas ya 

10 atendidas de ese mismo procedimiento (y odontólogo, si hay 

11 suficientes registros). 

12 3. CATÁLOGO — si tampoco hay historial, se usa la duración promedio 

13 declarada en el catálogo. 

14 

15La estimación mejora sola conforme la clínica acumula citas atendidas. 

16""" 

17import os 

18from dataclasses import dataclass 

19 

20from django.db.models import Avg, Count 

21 

22DIAS = ['LUN', 'MAR', 'MIE', 'JUE', 'VIE', 'SAB', 'DOM'] 

23 

24MINIMO_REGISTROS_HISTORICOS = 3 # por debajo de esto el promedio no es fiable 

25REDONDEO_MIN = 5 # las citas se agendan en múltiplos de 5 minutos

28@dataclass 

29class Estimacion: 

30 """Resultado de una estimación de duración.""" 

31 duracion_min: int 

32 bloque_min: int 

33 bloque_max: int 

34 origen: str # 'modelo' | 'historico' | 'catalogo' 

35 detalle: str # texto explicativo para mostrar al usuario 

36 modelo = None # instancia de ModeloVersion cuando el origen es 'modelo'

38 @property 

39 def etiqueta_origen(self): 

40 return { 

41 'modelo': 'Modelo predictivo', 

42 'historico': 'Promedio histórico de la clínica', 

43 'catalogo': 'Duración de referencia del catálogo', 

44 }.get(self.origen, self.origen)

47def _redondear(minutos): 

48 """Redondea al múltiplo de 5 más cercano, con un mínimo de 10 minutos.""" 

49 return max(10, int(round(minutos / REDONDEO_MIN) * REDONDEO_MIN))

52def _bloque_recomendado(duracion): 

53 """Un rango alrededor de la estimación: el mínimo para no quedarse corto y 

54 el máximo para no bloquear agenda de más.""" 

55 return _redondear(duracion * 0.85), _redondear(duracion * 1.20)

58def _modelo_activo(): 

59 from .models import ModeloVersion 

60 return ModeloVersion.objects.filter(activo=True, estado=True).first()

63# ──────────────────────────── caché del modelo ──────────────────────────── 

64# El archivo del modelo puede pesar varios megabytes. Deserializarlo en cada 

65# estimación significaría releerlo en cada cita que se agenda, de modo que se 

66# conserva en memoria y solo se recarga cuando cambia la versión activa. 

67_CACHE = {'version': None, 'estimador': None, 'anestesia': {}} 

68 

69 

70def limpiar_cache(): 

71 """Descarta el modelo en memoria. Se invoca al activar otra versión.""" 

72 _CACHE['version'] = None 

73 _CACHE['estimador'] = None 

74 _CACHE['anestesia'] = {}

77def _cargar(modelo): 

78 """Devuelve el estimador de la versión indicada, cargándolo una sola vez.""" 

79 if _CACHE['version'] == modelo.pk and _CACHE['estimador'] is not None: 

80 return _CACHE['estimador'] 

81 try: 

82 import joblib 

83 except ImportError: 

84 return None 

85 if not modelo.ruta_archivo or not os.path.exists(modelo.ruta_archivo): 

86 return None 

87 try: 

88 estimador = joblib.load(modelo.ruta_archivo) 

89 except Exception: 

90 # Un archivo dañado o entrenado con otra versión de la biblioteca no 

91 # debe impedir que se agende: la cadena de respaldo continúa. 

92 return None 

93 _CACHE['version'] = modelo.pk 

94 _CACHE['estimador'] = estimador 

95 return estimador

98def _anestesia_habitual(procedimiento): 

99 """Proporción histórica de atenciones de ese procedimiento con anestesia. 

100 

101 En el momento de agendar no se sabe si se aplicará anestesia, de modo que se 

102 usa el comportamiento habitual del procedimiento. La complicación, en 

103 cambio, se asume ausente: por definición es imprevista, y el sistema estima 

104 el caso nominal. 

105 """ 

106 if procedimiento.pk in _CACHE['anestesia']: 

107 return _CACHE['anestesia'][procedimiento.pk] 

108 from clinico.models import NotaClinica 

109 notas = NotaClinica.objects.filter(cita__procedimiento=procedimiento, 

110 estado=True) 

111 total = notas.count() 

112 valor = 1 if total and notas.filter(anestesia=True).count() * 2 >= total else 0 

113 _CACHE['anestesia'][procedimiento.pk] = valor 

114 return valor

117def _estimar_con_modelo(modelo, procedimiento, odontologo, fecha, hora, num_piezas): 

118 """Carga el modelo entrenado y predice la duración. 

119 

120 Mientras el archivo serializado no exista o las bibliotecas de aprendizaje 

121 automático no estén instaladas en el servidor, devuelve None para que la 

122 cadena de respaldo continúe con el promedio histórico. 

123 """ 

124 estimador = _cargar(modelo) 

125 if estimador is None: 

126 return None 

127 try: 

128 import pandas as pd 

129 except ImportError: 

130 return None 

131 

132 try: 

133 # Los nombres de las columnas deben coincidir con los del entrenamiento. 

134 entrada = pd.DataFrame([{ 

135 'procedimiento': procedimiento.nombre, 

136 'odontologo': f'{odontologo.nombres} {odontologo.apellidos}', 

137 'dia_semana': DIAS[fecha.weekday()], 

138 'hora_inicio': hora.hour, 

139 'num_piezas': num_piezas, 

140 'anestesia': _anestesia_habitual(procedimiento), 

141 'complicacion': 0, 

142 }]) 

143 return float(estimador.predict(entrada)[0]) 

144 except Exception: 

145 # Un fallo del modelo no debe impedir agendar una cita. 

146 return None

149def _estimar_con_historico(procedimiento, odontologo): 

150 """Promedio de la duración real de las citas atendidas de ese procedimiento. 

151 

152 Si hay suficientes registros del mismo odontólogo se prefiere ese promedio, 

153 porque el ritmo de trabajo varía entre profesionales. 

154 """ 

155 from agenda.models import Cita 

156 

157 base = Cita.objects.filter( 

158 procedimiento=procedimiento, 

159 estado_cita='ATENDIDA', 

160 duracion_real_min__isnull=False, 

161 estado=True, 

162 ) 

163 

164 del_odontologo = base.filter(odontologo=odontologo).aggregate( 

165 promedio=Avg('duracion_real_min'), total=Count('id')) 

166 if (del_odontologo['total'] or 0) >= MINIMO_REGISTROS_HISTORICOS: 

167 return del_odontologo['promedio'], del_odontologo['total'], True 

168 

169 general = base.aggregate(promedio=Avg('duracion_real_min'), total=Count('id')) 

170 if (general['total'] or 0) >= MINIMO_REGISTROS_HISTORICOS: 

171 return general['promedio'], general['total'], False 

172 

173 return None, 0, False

176def estimar_duracion(procedimiento, odontologo, fecha, hora, num_piezas=1): 

177 """Estima la duración de un procedimiento y el bloque de tiempo sugerido.""" 

178 # --- Nivel 1: modelo predictivo entrenado --- 

179 modelo = _modelo_activo() 

180 if modelo: 

181 prediccion = _estimar_con_modelo(modelo, procedimiento, odontologo, 

182 fecha, hora, num_piezas) 

183 if prediccion: 

184 duracion = _redondear(prediccion) 

185 minimo, maximo = _bloque_recomendado(duracion) 

186 est = Estimacion(duracion, minimo, maximo, 'modelo', 

187 f'Estimado por {modelo.nombre} ' 

188 f'({modelo.get_algoritmo_display()}).') 

189 est.modelo = modelo 

190 return est 

191 

192 # --- Nivel 2: promedio histórico real de la clínica --- 

193 promedio, total, es_del_odontologo = _estimar_con_historico(procedimiento, odontologo) 

194 if promedio: 

195 duracion = _redondear(promedio) 

196 minimo, maximo = _bloque_recomendado(duracion) 

197 alcance = ('de este odontólogo' if es_del_odontologo 

198 else 'de todos los odontólogos') 

199 return Estimacion(duracion, minimo, maximo, 'historico', 

200 f'Promedio de {total} atenciones previas {alcance}. ' 

201 f'La precisión aumentará al acumularse más registros.') 

202 

203 # --- Nivel 3: duración de referencia del catálogo --- 

204 duracion = _redondear(procedimiento.duracion_promedio_min) 

205 minimo, maximo = _bloque_recomendado(duracion) 

206 return Estimacion(duracion, minimo, maximo, 'catalogo', 

207 'Aún no hay suficientes atenciones registradas de este ' 

208 'procedimiento; se usa la duración de referencia del catálogo.')
