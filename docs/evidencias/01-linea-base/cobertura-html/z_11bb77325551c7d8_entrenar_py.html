1"""Entrena una versión del modelo de estimación de duraciones. 

2 

3Se ejecuta fuera del servidor web, normalmente desde una tarea programada: 

4 

5 python manage.py entrenar --algoritmo RF --activar 

6 

7Mantenerlo aquí y no en una vista evita cargar scikit-learn y pandas en los 

8procesos que atienden peticiones, que es el costo real de este componente: 

9alrededor de 170 MB de memoria por proceso frente a menos de un segundo de 

10cómputo. 

11""" 

12from django.core.management.base import BaseCommand 

13 

14from inteligencia import entrenamiento 

15 

16 

17class Command(BaseCommand): 

18 help = ('Entrena el modelo de estimación de duraciones y registra la ' 

19 'versión resultante con sus métricas.') 

20 

21 def add_arguments(self, parser): 

22 parser.add_argument('--algoritmo', default='RF', 

23 choices=sorted(entrenamiento.ALGORITMOS), 

24 help='Algoritmo a entrenar (por omisión, RF).') 

25 parser.add_argument('--activar', action='store_true', 

26 help='Pone la versión entrenada en producción.') 

27 parser.add_argument('--solo-si-mejora', action='store_true', 

28 help='Activa la versión únicamente si su error ' 

29 'absoluto medio es menor que el de la versión ' 

30 'actualmente en producción.') 

31 

32 def handle(self, *args, **opciones): 

33 algoritmo = opciones['algoritmo'] 

34 

35 altas, cambios, bajas = entrenamiento.sincronizar() 

36 self.stdout.write( 

37 f'Conjunto de entrenamiento: {altas} altas, {cambios} ' 

38 f'actualizaciones, {bajas} bajas.') 

39 

40 self.stdout.write(f'Entrenando {entrenamiento.ALGORITMOS[algoritmo]}…') 

41 

42 resultado = entrenamiento.entrenar(algoritmo) 

43 if resultado is None: 

44 self.stdout.write(self.style.WARNING( 

45 f'No hay registros suficientes: se requieren al menos ' 

46 f'{entrenamiento.MINIMO_REGISTROS} atenciones con duración real. ' 

47 f'La estimación sigue recayendo en el promedio histórico.')) 

48 return 

49 

50 self.stdout.write( 

51 f' Registros {resultado.registros} ' 

52 f'({resultado.registros_prueba} reservados para evaluación)\n' 

53 f' Entrenamiento {resultado.segundos * 1000:.0f} ms\n' 

54 f' Error absoluto medio {resultado.mae} min\n' 

55 f' RMSE {resultado.rmse} min\n' 

56 f' R² {resultado.r2}\n' 

57 f' Línea base {resultado.mae_base} min ' 

58 f'(estimar siempre el promedio)\n' 

59 f' Mejora {resultado.mejora_pct}%\n' 

60 f' Archivo {resultado.tamano_kb} KB') 

61 

62 meta = self.style.SUCCESS if resultado.cumple_meta else self.style.WARNING 

63 self.stdout.write(meta( 

64 f' Catálogo {resultado.mae_catalogo} min ' 

65 f'(reservar el bloque de referencia)\n' 

66 f' Reducción {resultado.mejora_catalogo_pct}% frente al ' 

67 f'catálogo; la meta del proyecto es ' 

68 f'{entrenamiento.META_REDUCCION_PCT:.0f}%.')) 

69 

70 from inteligencia.models import ModeloVersion 

71 activa = ModeloVersion.objects.filter(activo=True, estado=True).first() 

72 

73 activar = opciones['activar'] 

74 if opciones['solo_si_mejora']: 

75 if activa and activa.mae is not None and resultado.mae >= activa.mae: 

76 activar = False 

77 self.stdout.write(self.style.WARNING( 

78 f' La versión en producción tiene un error de {activa.mae} min; ' 

79 f'la nueva no mejora y se conserva como histórico.')) 

80 else: 

81 activar = True 

82 

83 version = entrenamiento.registrar_version(resultado, activar=activar) 

84 estado = 'activada en producción' if activar else 'registrada como histórico' 

85 self.stdout.write(self.style.SUCCESS( 

86 f'Versión «{version.nombre}» {estado}.')) 

87 

88 if resultado.importancias: 

89 self.stdout.write('\n Variables de mayor peso:') 

90 for i in resultado.importancias[:5]: 

91 self.stdout.write(f' {i["peso"]:5.1f}% {i["variable"]}')
