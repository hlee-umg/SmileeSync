382 def sembrar_agenda(self, lee, giron, pacientes, procedimientos): 

383 """Genera dos meses de agenda: historia atendida y citas por atender. 

384 

385 La historia es la que alimenta al modelo predictivo, de modo que no se 

386 construye con duraciones iguales a las del catálogo: la duración real de 

387 cada atención se obtiene de un procedimiento base afectado por el ritmo 

388 del profesional, el número de piezas tratadas y la aparición de 

389 complicaciones, más una dispersión aleatoria. Así el promedio histórico 

390 y el modelo tienen señal real que recuperar. 

391 """ 

392 from django.utils import timezone

394 from agenda.models import Cita 

395 from agenda.disponibilidad import asignar_unidad, franjas_efectivas

397 rng = random.Random(self.SEMILLA) 

398 hoy = timezone.localdate() 

399 lunes = hoy - timedelta(days=hoy.weekday()) 

400 inicio = lunes - timedelta(weeks=self.SEMANAS_ATRAS) 

401 fin = lunes + timedelta(weeks=self.SEMANAS_ADELANTE, days=4) 

402 

403 lista_pacientes = list(pacientes.values()) 

404 odontologos = [lee, giron] 

405 # Pacientes en tratamiento de ortodoncia: vuelven cada mes a control.

408 def elegir_procedimiento(odo):

410 codigos = [c for c, _ in opciones] 

411 pesos = [p for _, p in opciones] 

412 return procedimientos[rng.choices(codigos, weights=pesos)[0]]

419 if codigo == 'PROC-010': 

420 return rng.choice([1, 2]) 

421 return 1
