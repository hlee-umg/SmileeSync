21 # Los campos 'puede_*' conceden acceso al módulo; este determina si ese 

22 # acceso incluye escritura. Un rol de solo consulta ve la información de 

23 # sus módulos, pero no puede crearla ni modificarla. 

24 solo_lectura = models.BooleanField(

28 # El expediente clínico es un acto profesional, no una tarea 

29 # administrativa: quien lo escribe es el odontólogo, aunque su rol sea de 

30 # solo consulta para la agenda, y no lo escribe la recepción, aunque sí 

31 # modifique otros módulos. Por eso la escritura del expediente se declara 

32 # aparte en lugar de derivarse de `solo_lectura`.

54 una pregunta secreta para la recuperación de contraseña."""

79 def es_admin(self):

82 def puede(self, modulo): 

83 """Devuelve True si el usuario tiene acceso al módulo indicado.""" 

84 if self.es_admin: 

85 return True 

86 return bool(self.rol and getattr(self.rol, f'puede_{modulo}', False)) 

87 

88 @property 

89 def es_solo_consulta(self): 

90 """Un rol de solo consulta no modifica información en ningún módulo.""" 

91 if self.es_admin: 

92 return False 

93 return bool(self.rol and self.rol.solo_lectura) 

94 

95 def puede_escribir(self, modulo): 

96 """Acceso de escritura: exige el módulo y que el rol no sea de consulta. 

97 

98 El módulo clínico es la excepción y se rige por `escritura_clinica`: el 

99 expediente lo escribe quien tiene competencia profesional declarada, 

100 con independencia de sus privilegios administrativos. 

101 """ 

102 if self.es_admin: 

103 return True 

104 if not self.puede(modulo): 

105 return False 

106 if modulo == 'clinico': 

107 return bool(self.rol and self.rol.escritura_clinica) 

108 return not self.es_solo_consulta

110 def set_respuesta_secreta(self, texto): 

111 """Guarda la respuesta secreta cifrada (normalizada a minúsculas).""" 

112 self.respuesta_secreta = make_password((texto or '').strip().lower()) 

113 

114 def check_respuesta_secreta(self, texto): 

115 """Compara una respuesta con la guardada.""" 

116 if not self.respuesta_secreta: 

117 return False 

118 return check_password((texto or '').strip().lower(), self.respuesta_secreta)
